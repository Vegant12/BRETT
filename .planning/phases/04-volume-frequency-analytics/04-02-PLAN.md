```python
    """Shared config for every analytics return type.

    ``hide_input_in_errors`` for the same reason ``hevy/schemas.py`` sets it: a ValidationError
    would otherwise render ``input_value=...`` -- weekly tonnage, workout counts, body-composition

    somebody else's wire format. An unexpected field here is a bug in this repo, and it should
    fail loudly rather than be dropped. ``frozen=True`` because this is a read path -- a result
    object that a caller can mutate is a result object two callers will disagree about.
    """
```

```python
class FreshnessMetadata(_AnalyticsModel):
    """DQ-01. Per-resource ``last_synced_at``, keyed by the ``RESOURCE_*`` constants.

    A per-resource breakdown, not one collapsed timestamp: ``get_muscle_group_volume`` depends
    on BOTH ``workouts`` and ``exercise_templates``, and a stale template sync silently changes
    every per-muscle number. The two do drift -- 25 minutes apart on the live database. A single
    "oldest of the relevant resources" scalar is a lossy projection Phase 6 can derive in one
    line; the reverse is impossible.

    Carries ``last_synced_at`` ONLY. Never ``sync_state.last_error``, ``status``, ``cursor_page``
    or ``cursor_timestamp`` -- MCP-06 and CLAUDE.md forbid internal tokens or error text in
    anything Claude can read, and ``format_sync_error`` exists precisely to bound that leak.
    """

    """DQ-02. Exactly D-06's three fields -- do not add a fourth.

    ``earliest_data_at`` is what lets Claude tell "you did not train in this window" apart from
    "your synced history does not reach back that far". ``requested_window`` was explicitly
    considered and NOT selected; do not echo the range back here.

class MuscleGroupVolumeResult(_AnalyticsModel):

class MuscleGroupFrequency(_AnalyticsModel):

    sessions: int

    insufficient_data: bool


class ExerciseFrequency(_AnalyticsModel):

    exercise_template_external_id: str
    title: str | None
    sessions: int
    sessions_per_week: Decimal
    insufficient_data: bool
```

```python
import re
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[2]
ANALYTICS = REPO_ROOT / "backend" / "app" / "analytics"

# Anchored at the START of a line (modulo indentation): a mention in a docstring or comment --
# of which there are several, deliberately -- must not be read as an import.
_INTEGRATIONS_IMPORT = re.compile(r"^\s*(?:import|from)\s+app\.integrations", re.MULTILINE)
_MCP_IMPORT = re.compile(r"^\s*(?:import|from)\s+app\.mcp", re.MULTILINE)

# The ORM class identifiers, CamelCase. Prose in this package refers to the tables in lowercase
# (`exercise_muscles`), so the class name appearing at all means somebody is querying it.
_MUSCLE_TABLE_CLASS = re.compile(r"\bExerciseMuscle(?:Override)?\b")

# S608 belt-and-braces: no hand-built SQL string may reach the driver from this package. The
# timezone is the only user-controlled value that touches SQL, and it must stay a BOUND
# parameter via `func.timezone(tz, col)`.
_RAW_SQL_TEXT = re.compile(r"(?<![\w.])(?:sa\.)?text\s*\(")

# Read-only in v1 (MCP-05 inherits this). `session_ids.add(...)` on a Python set is fine and
# deliberately not matched -- only session-bound mutations are.
_SESSION_WRITE = re.compile(r"\bsession\.(?:commit|flush|add|add_all)\s*\(")
_WRITE_IMPORT = re.compile(
    r"^\s*from\s+sqlalchemy\s+import\s+.*\b(?:insert|update|delete)\b", re.MULTILINE
)


def _analytics_modules() -> list[Path]:
    """Every committed module under `app/analytics/`.

    `rglob` rather than a hand-maintained list: a new analytics submodule must be covered by
    these boundaries the moment it exists, without anyone remembering to add it here.
    """
    return sorted(ANALYTICS.rglob("*.py"))
```

MODULE DOCSTRING must state, in the house voice, the four things a reader has to know:
(1) BODY-05 is a two-point latest-vs-earliest comparison over a fixed 90-day window and
deliberately NOT `bodyweight.py`'s rolling-average shape (D-05/D-06), with the live-data reason:
13 of the 16 in-scope columns have never been recorded, and `waist_cm` — the best-covered — holds
2 readings in the trailing 90 days, so a 3/5/10 measured-day floor would answer
`insufficient_data` to every real call. (2) `insufficient_data` and `is_stale` are INDEPENDENT
signals (D-08) and all four combinations are reachable. (3) There is no quantization here: the
source columns are `Numeric(6, 2)`/`Numeric(5, 2)`, so a `Decimal` subtraction of two 2-decimal
values is already exact and `bodyweight.py`'s `_WEIGHT_QUANTUM` has nothing to correct. (4) There
is no source-precedence tie-break: D-03 leaves `bodyweight.py`'s manual-wins rule dormant because
only `'hevy'` writes this table, so same-day rows are resolved by a deterministic ORDER BY rather
than by a precedence rule nobody chose.

    # D-06. Wide enough to usually catch two readings of an occasionally-tracked metric -- live
    # `waist_cm` has a 14-day median gap and a 119-day worst case -- while still reading as a
    # recent trend rather than an all-time comparison. INCLUSIVE on both ends:
    # [anchor - 89, anchor] is 90 calendar days, not 91.
    WINDOW_DAYS = 90

    # D-08. [ASSUMED], and echoed on every response as `stale_after_days` rather than hidden
    # here, precisely because no external authority governs "how old is too old" for a tape
    # measure. Set at roughly 2x the observed 14-day median gap between real `waist_cm` readings,
    # so a metric tracked at its historical cadence does not spuriously flag while one that has
    # gone quiet for a third of the window does. A one-constant change if the user disagrees.
    STALE_AFTER_DAYS = 30

behind Docker — the same split `bodyweight.py` makes with `_dedupe_by_day`/`_window_average`:

    def fold_by_day(readings: Sequence[MetricReading]) -> dict[dt.date, Decimal]:
        """One value per measured day.

        No precedence rule, deliberately (D-03). `bodyweight.py` prefers `'manual'` over `'hevy'`
        because BODY-01 has a manual-correction path; BODY-05 does not and will not under this
        phase's scope, so every row here is `source='hevy'` and the conflict this would resolve
        cannot occur on real data. The caller orders by `(measured_on, source)`, so the surviving
        value for a day is the alphabetically-last source -- DETERMINISTIC rather than arbitrary,
        which is the only property that matters when the tie cannot happen.
        """
        return {reading.measured_on: reading.value for reading in readings}


    def staleness(
        latest_measured_on: dt.date | None, *, anchor: dt.date
    ) -> tuple[bool, int | None]:
        """D-08: how old the newest reading is, computed INDEPENDENTLY of the coverage floor.

        Returns `(is_stale, days_since_latest_reading)`. With no reading in the window there is
        no "latest" to be stale, so the answer is `(False, None)` -- NOT `(True, None)`. "We have
        nothing" is `coverage.insufficient_data`'s sentence to say, and saying it twice in two
        fields would make a metric that was never measured indistinguishable from one that was
        abandoned.

        A single reading in the window still gets a real age. It failed the floor for a
        COMPARISON, which says nothing about whether the observation itself is recent.
        """
        if latest_measured_on is None:
            return False, None
        elapsed = (anchor - latest_measured_on).days
        return elapsed > STALE_AFTER_DAYS, elapsed

A PRIVATE QUERY HELPER modelled on `_weight_rows`, parameterized on the column:

    async def _metric_readings(
        session: AsyncSession,
        *,
        user_id: uuid.UUID,
        column: InstrumentedAttribute[Decimal | None],
        start_day: dt.date,
        end_day: dt.date,
    ) -> list[MetricReading]:
        """Every non-null reading of one metric in an INCLUSIVE local-date range.

        `column.is_not(None)` is in the WHERE clause rather than filtered in Python because a
        NULL is not a reading of zero -- Hevy writes all 17 columns on every sync and leaves the
        untracked ones NULL, so without this predicate `measured_days_in_window` would count
        every sync date and the floor would clear for a metric that was never measured.

        `measured_on` is a calendar Date compared against two `dt.date` bounds with no timezone
        conversion of any kind. Threading UTC instants through here by analogy with the workout
        aggregates would be a category error and would shift the boundary days by seven hours.

        Ordered `(measured_on, source)` so `fold_by_day`'s same-day resolution is
        order-INDEPENDENT rather than order-dependent by luck.
        """
        result = await session.execute(
            select(BodyMeasurement.measured_on, BodyMeasurement.source, column)
            .where(
                # T-11-02 / ASVS V4: the tenant predicate leads.
                BodyMeasurement.user_id == user_id,
                column.is_not(None),
                BodyMeasurement.measured_on >= start_day,
                BodyMeasurement.measured_on <= end_day,
            )
            .order_by(BodyMeasurement.measured_on, BodyMeasurement.source)
        )
        return [
            MetricReading(measured_on=row[0], source=row[1], value=row[2])
            for row in result.all()
        ]

THE PUBLIC FUNCTION:

    async def get_body_measurement_trend(
        session: AsyncSession, *, user_id: uuid.UUID, metric: BodyMeasurementMetric
    ) -> BodyMeasurementTrendResult:
        """BODY-05: one metric's latest reading against its earliest inside a fixed 90-day window.

        Takes no date range, for the same reason `get_body_weight_trend` does not: "how has my
        waist changed" is always a question about now, and a caller-supplied anchor would let a
        stale window produce a confident-looking delta that has since been superseded.

        `metric` is a `BodyMeasurementMetric`, never a string. `getattr` on a closed 16-member
        enum whose values are asserted equal to the ORM's own column names is the entire
        validation story (D-09) -- the MCP layer's schema rejects an unknown name before this
        function is reached, and a hand-written membership check here would be a second copy of
        the same list, free to drift from the first.

        `user_id` is keyword-only with no default (ASVS V4). The caller owns the transaction;
        nothing here writes.
        """
        column: InstrumentedAttribute[Decimal | None] = getattr(BodyMeasurement, metric.value)

        probe_day = local_day(utc_now(), "UTC")
        tz = (
            await resolve_window(
                session,
                user_id=user_id,
                start_day=probe_day,
                end_day=probe_day,
                snap_to_weeks=False,
            )
        ).tz
        anchor = local_day(utc_now(), tz)

        readings = await _metric_readings(
            session,
            user_id=user_id,
            column=column,
            start_day=anchor - dt.timedelta(days=WINDOW_DAYS - 1),
            end_day=anchor,
        )
        by_day = fold_by_day(readings)
        days = sorted(by_day)

        coverage = build_measurement_coverage(
            window_days=WINDOW_DAYS,
            measured_days=len(by_day),
            earliest_measurement_at=await earliest_metric_measurement_at(
                session, user_id=user_id, column=column
            ),
        )

        latest_day = days[-1] if days else None
        is_stale, days_since = staleness(latest_day, anchor=anchor)

        # The earliest endpoint and the delta exist ONLY above the floor. With one reading,
        # "earliest" IS "latest" and the delta is a guaranteed zero -- which reads as "your waist
        # has not changed" rather than "you measured it once", the single most misleading thing
        # this function could return.
        earliest_day = days[0] if not coverage.insufficient_data else None
        latest_value = by_day[latest_day] if latest_day is not None else None
        earliest_value = by_day[earliest_day] if earliest_day is not None else None
        delta = (
            latest_value - earliest_value
            if latest_value is not None and earliest_value is not None
            else None
        )

        return BodyMeasurementTrendResult(
            metric=metric,
            window_days=WINDOW_DAYS,
            anchor_day=anchor,
            coverage=coverage,
            # Populated whenever ANY reading exists, floor notwithstanding: a single reading is a
            # real observation, and suppressing it would report "no data" for a metric the user
            # demonstrably measured.
            latest_value=latest_value,
            latest_measured_on=latest_day,
            earliest_value=earliest_value,
            earliest_measured_on=earliest_day,
            # No `.quantize(...)`. Both operands are already 2-decimal `Numeric` values, so the
            # difference is exact -- `bodyweight.py`'s `_WEIGHT_QUANTUM` exists because it
            # AVERAGES, and this never does.
            delta_value=delta,
            is_stale=is_stale,
            days_since_latest_reading=days_since,
            stale_after_days=STALE_AFTER_DAYS,
            freshness=await build_freshness(
                session, user_id=user_id, resources=FRESHNESS_RESOURCES_BODY
            ),
        )

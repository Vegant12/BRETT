**`backend/app/mcp/context.py`:**
```python
"""Lifespan context handed to every MCP tool (MCP-01).

The context BORROWS ``app.state.sessionmaker`` and must never call
``make_engine()``. A second engine here would bind its own pool to this
lifespan's event loop, double the connection count against a free-tier Postgres
that allows very few, and silently diverge from the engine ``/health`` probes --
so the health check could report ``connected`` while every tool call failed.

``from app.main import app`` is deliberately inside the function body: ``main``
imports ``app.mcp.server``, which imports this module, so a module-scope import
would be circular. Deferring it to startup breaks the cycle, and it is safe
because ``mcp_lifespan`` runs once at startup -- after the FastAPI lifespan has
already assigned ``app.state.sessionmaker`` -- not per request.
"""

from collections.abc import AsyncIterator
from contextlib import asynccontextmanager
from dataclasses import dataclass

import structlog
from mcp.server import MCPServer
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker

log = structlog.get_logger()


@dataclass(slots=True)
class AppContext:
    """What every MCP tool is handed. Borrowed, never owned."""

    sessionmaker: async_sessionmaker[AsyncSession]


@asynccontextmanager
async def mcp_lifespan(server: MCPServer) -> AsyncIterator[AppContext]:
    """Yield the tool context, sourced from the FastAPI app's state."""
    from app.main import app

    log.info("mcp_lifespan_starting")
    try:
        yield AppContext(sessionmaker=app.state.sessionmaker)
    finally:
        log.info("mcp_lifespan_stopped")
```

```python
"""Database connectivity tool (D-08, MCP-01).

Throwaway in scope, permanent in shape: Phase 6 replaces this tool but not the
path it proves. A bare echo tool would show only that the transport responds; a
real ``SELECT 1`` through ``ctx.request_context.lifespan_context.sessionmaker``
proves the lifespan-wired session factory that every real analytics tool depends
on.

``ctx`` MUST be annotated ``Context[AppContext]``. With no annotation the SDK
exposes it to the model as a required string argument (``"properties": {"ctx":
{"type": "string"}}``) and every call fails schema validation.
"""

import structlog
from mcp.server.mcpserver import Context
from pydantic import BaseModel
from sqlalchemy import text

from app.mcp.context import AppContext

log = structlog.get_logger()


class PingResult(BaseModel):
    """Returning a BaseModel auto-populates outputSchema and structuredContent."""

    status: str
    database: str


async def fitness_ping(ctx: Context[AppContext]) -> PingResult:
    """Round-trip the database through the lifespan-wired session factory."""
    database = "connected"
    try:
        sessionmaker = ctx.request_context.lifespan_context.sessionmaker
        async with sessionmaker() as session:
            await session.execute(text("SELECT 1"))
    except Exception:  # noqa: BLE001 - any failure to reach the DB must degrade, not error
        # Mirrors health.py: the detail goes to the log (which runs through the
        # redaction pipeline); the caller gets only the coarse state. Connection
        # errors carry the DSN, host, and driver internals (T-02-09, FOUND-05).
        log.exception("mcp_ping_db_check_failed")
        database = "error"
    return PingResult(status="ok", database=database)
```

Create **`backend/app/mcp/auth.py`**:

```python
"""MCP bearer-token verification (MCP-04).

Two staged mechanisms behind one selector, so ``server.py`` never changes when
the mechanism does:

* ``static``  -- constant-time compare against ``MCP_AUTH_SECRET``. Proves the
  deploy + mount + auth pipeline through Claude Code's ``--header`` flag (D-01).
* ``authkit`` -- WorkOS AuthKit JWKS validation (D-03/D-04). Landed in plan
  02-06; this module raises rather than silently degrading to ``static``.

Nothing here may log a token, a decoded claim, or an exception body. A rejected
credential is still a credential: ``verify_token`` fails CLOSED and silent
(T-02-04). Comparison is ``secrets.compare_digest`` and never ``==``, because a
short-circuiting comparison leaks the secret one byte at a time (T-02-05).
"""

import secrets

from mcp.server.auth.provider import AccessToken, TokenVerifier

from app.config import Settings


class StaticTokenVerifier:
    """Constant-time comparison against ``MCP_AUTH_SECRET`` (D-01)."""

    def __init__(self, secret: str) -> None:
        self._secret = secret

    async def verify_token(self, token: str) -> AccessToken | None:
        if secrets.compare_digest(token, self._secret):
            return AccessToken(token=token, client_id="brett-cli", scopes=[])
        return None


def build_token_verifier(settings: Settings) -> TokenVerifier:
    """Select the verifier for the configured auth mode.

    Fails fast at import time rather than at first request: a server that starts
    without a working verifier is a server that is unauthenticated in production
    until someone notices.
    """
    if settings.mcp_auth_mode == "authkit":
        raise RuntimeError(
            "MCP_AUTH_MODE=authkit is not implemented until plan 02-06"
        )
    secret = settings.mcp_auth_secret
    if secret is None or not secret.get_secret_value():
        raise RuntimeError("MCP_AUTH_SECRET is required to start the MCP server")
    return StaticTokenVerifier(secret.get_secret_value())
```

Create **`backend/app/mcp/server.py`**:

```python
"""MCP server assembly (MCP-01, MCP-04, D-05).

Import order here is load-bearing in the same way ``main.py``'s is.
``streamable_http_app()`` MUST run at import: ``mcp.session_manager`` raises
``RuntimeError`` until it has, and ``main.py``'s lifespan reads that attribute.

Three non-obvious constraints, each verified against mcp 2.0.0:

* ``AuthSettings`` is mandatory alongside ``token_verifier``. Passing a verifier
  without it raises ``ValueError`` at import, and ``AuthSettings`` itself requires
  BOTH ``issuer_url`` and ``resource_server_url`` (RESEARCH Pitfall 4). During
  the static-bearer stage ``issuer_url`` is never exercised -- a valid header
  means no 401, so no discovery -- but it must still be a valid URL.
* ``TransportSecuritySettings`` is not optional in practice. Left at ``None`` the
  SDK auto-enables DNS-rebinding protection allowing only localhost, and every
  request through Render's real Host header returns ``421 Invalid Host header``
  with no MCP-level error body (T-02-01, RESEARCH Pitfall 2).
* ``allowed_origins=[]`` on purpose. Claude's connectors call server-to-server
  from Anthropic's egress range and send no ``Origin``; a non-empty list here
  creates a 403 trap for no benefit.

The sub-app returned below owns ``/mcp`` AND
``/.well-known/oauth-protected-resource/mcp`` at its OWN root, so ``main.py``
mounts it at ``/``. Mounting at ``/mcp`` relocates the metadata document to a
path Claude never probes and breaks OAuth discovery silently -- the MCP server
receives the first request while the authorization server sees no traffic at all
(RESEARCH Pitfall 3).
"""

from mcp.server import MCPServer
from mcp.server.auth.settings import AuthSettings
from mcp.server.transport_security import TransportSecuritySettings

from app.config import get_settings
from app.mcp.auth import build_token_verifier
from app.mcp.context import mcp_lifespan
from app.mcp.tools.ping import fitness_ping

settings = get_settings()  # lru_cached -- not a second read

mcp = MCPServer(
    name="BRETT",
    version="0.1.0",
    lifespan=mcp_lifespan,
    token_verifier=build_token_verifier(settings),
    auth=AuthSettings(
        issuer_url=settings.mcp_issuer_url,
        # Must equal the URL the user types into Claude, character for
        # character, and the Resource Indicator registered at the IdP
        # (RESEARCH Pitfall 8).
        resource_server_url=settings.mcp_resource_url,
        # Scope design belongs with the real tool set in Phase 6. An empty
        # scope list is the least likely thing to break a first handshake.
        required_scopes=None,
    ),
)

# D-08: one tool, and it does a real DB round-trip.
mcp.add_tool(
    fitness_ping,
    name="fitness_ping",
    description="Database connectivity check.",
)

# MUST run at import time. See the module docstring.
mcp_app = mcp.streamable_http_app(
    streamable_http_path="/mcp",
    # No session state to lose across a Render restart or spin-down; the
    # 2026-07-28 transport removed protocol-level sessions anyway.
    stateless_http=True,
    transport_security=TransportSecuritySettings(
        allowed_hosts=settings.mcp_allowed_hosts,
        allowed_origins=[],
    ),
)
```

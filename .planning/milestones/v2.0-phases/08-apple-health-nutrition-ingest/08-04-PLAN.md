```python
class HAEError(Exception):
    """Base for every Health Auto Export ingest failure."""
    safe_message: str = ""


class HAEPayloadError(HAEError):
    """The webhook body did not match the expected envelope.

    Wraps a pydantic ``ValidationError`` WITHOUT reproducing its message:
    ``ValidationError.__str__`` renders ``input_value=...``, i.e. this person's calorie counts
    and macro grams, into a string that would reach a log line and `sync_state.last_error`.
    Constructed from NOTHING, deliberately: an ``__init__`` that accepted a message would let a
    future caller interpolate the payload into it and no test elsewhere would notice.
    """
    def __init__(self) -> None:
        self.safe_message = "Health Auto Export payload did not match the expected envelope"
        super().__init__(self.safe_message)


class HAEDateFormatError(HAEError):
    """A datapoint timestamp matched none of the known formats.

    Carries NO value. A timestamp is not health data, but it is user-controlled input reaching a
    log sink, and there is nothing a caller can do with it that the format list does not already
    say.
    """
    def __init__(self) -> None:
        self.safe_message = "Health Auto Export timestamp matched no known format"
        super().__init__(self.safe_message)


class UnknownUnitError(HAEError):
    """A metric arrived in a unit the registry does not accept.

    Carries the METRIC NAME and the UNIT STRING and never the quantity: a metric name and a unit
    are structural facts, a qty is personal health data (Shared Pattern 3). The unit is truncated
    because it is attacker-controlled and lands in a log line.
    """
    _MAX_UNIT_LEN = 32

    def __init__(self, *, metric: str, unit: str) -> None:
        self.metric = metric
        self.unit = unit[: self._MAX_UNIT_LEN]
        self.safe_message = f"unsupported unit {self.unit!r} for metric {metric!r}"
        super().__init__(self.safe_message)
```

```python
Agg = Literal["sum", "avg"]
Kind = Literal["activity", "nutrition"]   # maps 1:1 to D-03's two sync_state resources


class MetricSpec(NamedTuple):
    canonical_unit: str
    accepted_units: frozenset[str]        # every unit HAE is known to emit for this metric
    aggregation: Agg
    kind: Kind


# The ONLY place a metric name appears. `MappingProxyType` for the reason `exercise_map.py`'s
# tables use it: a caller must not be able to retune ingestion at runtime.
#
# ### The naming trap ###
# HAE's `dietary_` prefix is INCONSISTENT and guessing it wrong is SILENT, not an error --
# D-04 drops an unlisted metric with a log line and no test failure. The vendor's TypeScript
# enum KEYS are uniformly `DIETARY_*` while the WIRE VALUES mostly are not:
#     carries the prefix : dietary_energy, dietary_sugar
#     does NOT carry it  : protein, carbohydrates, total_fat, fiber, sodium
# It is `total_fat`, never `fat`. It is `dietary_sugar`, never `sugar`. It is `protein`, never
# `dietary_protein` -- a real third-party fixture in the wild gets this wrong.
#
# Absent on purpose: `weight_body_mass` (D-01 -- no Apple Health body mass in Phase 8, and no
# third source on `body_measurements`) and anything workout-shaped (D-02).
METRIC_REGISTRY: Mapping[str, MetricSpec] = MappingProxyType({
    # --- Activity (HEALTH-06) ---
    "step_count":               MetricSpec("count", frozenset({"count"}),      "sum", "activity"),
    "active_energy":            MetricSpec("kcal",  frozenset({"kcal", "kJ"}), "sum", "activity"),
    "basal_energy_burned":      MetricSpec("kcal",  frozenset({"kcal", "kJ"}), "sum", "activity"),
    "walking_running_distance": MetricSpec("km",    frozenset({"km", "mi"}),   "sum", "activity"),
    "flights_climbed":          MetricSpec("count", frozenset({"count"}),      "sum", "activity"),

    # --- Nutrition (NUTR-02) ---
    "dietary_energy":           MetricSpec("kcal",  frozenset({"kcal", "kJ"}), "sum", "nutrition"),
    "protein":                  MetricSpec("g",     frozenset({"g"}),          "sum", "nutrition"),
    "carbohydrates":            MetricSpec("g",     frozenset({"g"}),          "sum", "nutrition"),
    "total_fat":                MetricSpec("g",     frozenset({"g"}),          "sum", "nutrition"),
    "fiber":                    MetricSpec("g",     frozenset({"g"}),          "sum", "nutrition"),
    "dietary_sugar":            MetricSpec("g",     frozenset({"g"}),          "sum", "nutrition"),
    "sodium":                   MetricSpec("mg",    frozenset({"mg", "g"}),    "sum", "nutrition"),
})

_KJ_PER_KCAL = Decimal("4.184")
_KM_PER_MILE = Decimal("1.609344")
_MG_PER_G = Decimal("1000")


def to_canonical(qty: Decimal, wire_unit: str, spec: MetricSpec, *, metric: str) -> Decimal:
    """Convert at the boundary. An UNKNOWN unit is a REJECTED DATAPOINT, never a raw number.

    HAE units are a per-metric user preference inside the phone app and are converted on-device
    at export time, so a settings change silently alters the wire units on the next export.
    Storing an unconverted value would be a plausible number that is quietly wrong -- this
    project's stated primary risk. An empty `wire_unit` (the metric-level `units` field was
    absent) reaches the same rejection, deliberately: assuming the canonical unit is exactly the
    silent failure this function exists to prevent.
    """
    if wire_unit not in spec.accepted_units:
        raise UnknownUnitError(metric=metric, unit=wire_unit)
    if wire_unit == spec.canonical_unit:
        return qty
    if spec.canonical_unit == "kcal" and wire_unit == "kJ":
        return qty / _KJ_PER_KCAL
    if spec.canonical_unit == "km" and wire_unit == "mi":
        return qty * _KM_PER_MILE
    if spec.canonical_unit == "mg" and wire_unit == "g":
        return qty * _MG_PER_G
    # Unreachable while `accepted_units` and the branches above agree; a KeyError-equivalent
    # rather than a silent `return qty`, which is how an unconverted value gets stored.
    raise UnknownUnitError(metric=metric, unit=wire_unit)
```

```python
class HAEDataPoint(_HAEModel):
    """One datapoint. EVERY field is optional, and that is a deliberate deviation.

    Health Auto Export has no documented retry, so a rejected batch is permanent data loss for the
    whole window (D-04). Pydantic validates a payload atomically, so one required field would let
    a single odd datapoint -- e.g. a heart-rate metric the user toggled on, whose datapoints carry
    Min/Max/Avg and no `qty` at all -- destroy an entire day of steps AND nutrition. Optional
    fields push the decision down to `normalize.py`, which drops ONE datapoint and logs it.

    `qty` is `Decimal`, never `float`: HAE prorates `count`-unit metrics across aggregation
    buckets, so `flights_climbed` really does arrive as 0.20163697817574894 and a float would
    make the same inputs produce order-dependent sums (RESEARCH Finding C).

    `date` is the RAW wire string, parsed by `parse_hae_datetime` in `normalize.py`. Parsing here
    would make one malformed timestamp a whole-payload failure, which is the same trap again.

    `source` is `str | None`: it is ABSENT ENTIRELY for gait metrics in every real captured
    payload, despite the vendor's own TypeScript interface declaring it required.
    """
```

```python
# Real payloads: "2025-08-15 00:00:00 +0900" -- NOT ISO 8601, which is why pydantic's native
# `datetime` field type is not used. The 12-hour variant appears in 12-hour locales, and newer
# iOS inserts U+202F NARROW NO-BREAK SPACE before AM/PM rather than a plain space. A third-party
# receiver additionally reports seeing a true ISO form.
#
# THIS PARSER IS SHARED WITH PHASE 9. `export.xml`, `records.csv` and the webhook JSON all use
# this format -- verified against the user's own records.csv ("2026-08-21 12:35:53 +0700"). Do
# not write a second one; two parsers produce two shapes and the historical trend silently
# disagrees with the ongoing one.
_HAE_DATE_FORMATS: tuple[str, ...] = (
    "%Y-%m-%d %H:%M:%S %z",
    "%Y-%m-%d %I:%M:%S %p %z",
)
# Written as an ESCAPE, never as a literal: a bare U+202F in source is invisible in every
# diff and every review, and one careless editor normalising it to a plain space silently
# disables this whole branch.
_NARROW_NO_BREAK_SPACE = "\u202f"


def parse_hae_datetime(value: str) -> dt.datetime:
    """Parse an HAE timestamp into a TIMEZONE-AWARE datetime. FOUND-04.

    Raises `HAEDateFormatError`, which carries no input, rather than returning None: a naive or
    unparsed timestamp silently mis-buckets a whole day (`core.time._require_aware` exists for
    exactly this reason).
    """
    candidate = value.strip().replace(_NARROW_NO_BREAK_SPACE, " ")
    for fmt in _HAE_DATE_FORMATS:
        try:
            parsed = dt.datetime.strptime(candidate, fmt)
        except ValueError:
            continue
        if parsed.tzinfo is None or parsed.tzinfo.utcoffset(parsed) is None:
            raise HAEDateFormatError
        return parsed
    try:
        parsed = dt.datetime.fromisoformat(candidate)
    except ValueError:
        raise HAEDateFormatError from None
    if parsed.tzinfo is None or parsed.tzinfo.utcoffset(parsed) is None:
        raise HAEDateFormatError
    return parsed
```

`raise ... from None` throughout: chaining would attach the original `ValueError`, whose message
renders the input string.

**A. `backend/app/integrations/health_auto_export/auth.py`**:

```python
"""Shared-secret verification for the Health Auto Export webhook (HEALTH-07).

Deliberately NOT an MCP `TokenVerifier`: importing `mcp.server.auth.provider` here would put an
MCP dependency inside an integrations package and break the boundary this package exists to
draw. The decision function returns a plain `bool`; the HTTP contract (a bare 401 that writes
nothing and explains nothing) is the route's job, in `app/api/routes/health_ingest.py`.
"""

_BEARER_PREFIX = "Bearer "


def verify_bearer(header_value: str | None, secret: str) -> bool:
    """Constant-time check of an ``Authorization: Bearer <token>`` header.

    Compared as BYTES, never as `str`. `secrets.compare_digest` raises `TypeError` on a `str`
    holding non-ASCII characters, and the ASGI server latin-1 decodes header bytes -- so a single
    arbitrary byte in the Authorization header would escape as an unhandled exception and surface
    as a 500 with a traceback rather than a silent 401, breaking this module's fail-closed
    contract. The bytes overload has no such restriction and is still constant-time.

    Fails closed on an empty secret: an unconfigured deployment has a DISABLED ingest path, never
    an open one.

    The scheme prefix is checked before the compare. That is not a leak: the presence of the word
    "Bearer" is not secret, and the secret itself is never involved in a short-circuit.
    """
    if not secret or header_value is None:
        return False
    if not header_value.startswith(_BEARER_PREFIX):
        return False
    presented = header_value[len(_BEARER_PREFIX) :]
    return secrets.compare_digest(presented.encode("utf-8"), secret.encode("utf-8"))
```
